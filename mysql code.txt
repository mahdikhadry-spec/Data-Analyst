WITH duplicate_rows AS (
    SELECT
        *,
        COUNT(*) OVER (
            PARTITION BY
                person_age,
                person_income,
                person_home_ownership,
                person_emp_length,
                loan_intent,
                loan_grade,
                loan_amnt,
                loan_int_rate,
                loan_status,
                loan_percent_income,
                cb_person_default_on_file,
                cb_person_cred_hist_length
        ) AS duplicate_count
    FROM loan_data
)
SELECT COUNT(*) AS total_rows_in_duplicate_groups
FROM duplicate_rows
WHERE duplicate_count > 1;

WITH ranked_rows AS (
    SELECT
        loan_id,
        ROW_NUMBER() OVER (
            PARTITION BY
                person_age,
                person_income,
                person_home_ownership,
                person_emp_length,
                loan_intent,
                loan_grade,
                loan_amnt,
                loan_int_rate,
                loan_status,
                loan_percent_income,
                cb_person_default_on_file,
                cb_person_cred_hist_length
            ORDER BY loan_id
        ) AS row_numbers
    FROM loan_data
)
SELECT COUNT(*) AS rows_to_remove
FROM ranked_rows
WHERE row_numbers > 1;

   sum(loan_status = 0) as non_defaulted_loans,

   from loan_data

   loan_intent,

   round(avg(loan_amnt)) as avg_loan_size,
   round(sum(loan_status = 1) / count(*) *100) as default_percentage
	from loan_data
   group by loan_intent
   order by round(sum(loan_status = 1) / count(*) *100) desc;

   round(sum(loan_status = 0) / count(*) *100) as non_default_percentage
   from loan_data
   group by loan_grade
   order by sum(loan_amnt) desc, round(sum(loan_status = 1) / count(*) *100) desc ;

   from loan_data;
   
   select Case
   When loan_percent_income < 0.10 then "Less than 10%"
   When loan_percent_income between 0.10 and 0.30 then "Between 10 to 30 percent"
   When loan_percent_income between 0.30 and 0.50 then "Between 30 to 50 percent"
   When loan_percent_income between 0.50 and 0.80 then "Between 50 to 80 percent"
   else "Over 80 Percent"

   count(*) as total_loans,
   sum(loan_status =1) as defaulted_loans, sum(loan_status=1) / count(*) *100 as defaul_percentage
   from loan_data

   When loan_percent_income < 0.10 then "Less than 10%"
   When loan_percent_income between 0.10 and 0.30 then "Between 10 to 30 percent"
   When loan_percent_income between 0.30 and 0.50 then "Between 30 to 50 percent"

   when loan_amnt between 15000 and 25000 then "Between 15000 and 25000"

-- calculate loss of revenue based on highest risk combinations

SELECT
    loan_grade,
    cb_person_default_on_file,
    person_home_ownership,
    COUNT(*) AS number_of_loans,
    round(
		sum(case
        when loan_status = 1 then loan_amnt 
        else 0
        end)
        ) as defaulted_loan_value,
    
    SUM(loan_status = 1) AS defaulted_loans,
    ROUND(SUM(loan_status = 1) / COUNT(*) * 100, 2)
        AS default_percentage

FROM loan_data
GROUP BY
    loan_grade,
    cb_person_default_on_file,
    person_home_ownership

HAVING COUNT(*) >= 20
ORDER BY defaulted_loan_value DESC
LIMIT 20;


SELECT
    loan_grade,

    SUM(
        CASE
		WHEN loan_status = 1 THEN loan_amnt
		ELSE 0
        END
    ) AS total_defaulted_loan_value,

    ROUND(
        SUM(
            CASE
			WHEN loan_status = 1 THEN loan_amnt
			ELSE 0
            END
        )
        /
        SUM(
            SUM(
                CASE
                    WHEN loan_status = 1 THEN loan_amnt
                    ELSE 0
                END
            )
        ) OVER () * 100,
        2
    ) AS share_of_total_defaulted_value

FROM loan_data
GROUP BY loan_grade
ORDER BY total_defaulted_loan_value DESC;
